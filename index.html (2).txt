    <script>
        // 語言狀態：'zh' 為中文，'en' 為英文
        let currentLang = 'zh';
        let currentPage = 0;

        // 繪本每一頁的資料集
        const storyData = [

                    { id: "orange", zh: "橘子", en: "Orange", icon: "🍊" }
                ],
                successZh: "答對了！香蕉甜甜的很好吃！",
                successEn: "Correct! Bananas are sweet and delicious!"
            },
            {
                zh: "最後，小熊想買一個大西瓜在夏天消暑。請指給牠看！",

                type: "outro"
            }
        ];

        // 渲染頁面
        function renderPage() {
            const page = storyData[currentPage];
            
            // 更新文字
            document.getElementById('story-content').innerText = page[currentLang];
            
            // 更新圖片/Emoji
            document.getElementById('image-area').innerHTML = `<div style="font-size: 8rem; margin: 20px 0;">${page.image}</div>`;
            
            // 更新互動區
            const interactiveBox = document.getElementById('interactive-box');
            interactiveBox.innerHTML = '';
            
            if (page.type === 'game') {
                page.options.forEach(opt => {
                    const btn = document.createElement('button');
                    btn.className = 'fruit-btn';
                    btn.innerHTML = `${opt.icon} <span>${currentLang === 'zh' ? opt.zh : opt.en}</span>`;
                    btn.onclick = () => checkAnswer(opt.id, page);
                    interactiveBox.appendChild(btn);
                });
            }

            // 控制導覽按鈕狀態
            document.getElementById('prev-btn').disabled = currentPage === 0;
            document.getElementById('next-btn').disabled = currentPage === storyData.length - 1;
        }

        // 檢查互動答案
        function checkAnswer(selectedId, page) {
            if (selectedId === page.target) {
                document.getElementById('story-content').innerText = page[`success${currentLang.charAt(0).toUpperCase() + currentLang.slice(1)}`];
                alert(currentLang === 'zh' ? "🎉 答對了！" : "🎉 Correct!");
            } else {
                alert(currentLang === 'zh' ? "❌ 再試一次看看！" : "❌ Try again!");
            }
        }

        // 切換中英文
        function toggleLanguage() {
            currentLang = currentLang === 'zh' ? 'en' : 'zh';
            document.getElementById('lang-btn').innerText = currentLang === 'zh' ? 'English' : '繁體中文';
            document.getElementById('main-title').innerText = currentLang === 'zh' ? '🐻 幫小熊買水果' : '🐻 Help Little Bear Buy Fruits';
            renderPage();
        }

        // 上下頁控制
        function changePage(direction) {
            currentPage += direction;
            if (currentPage < 0) currentPage = 0;
            if (currentPage >= storyData.length) currentPage = storyData.length - 1;
            renderPage();
        }

        // 瀏覽器內建語音朗讀 (TTS 功能)
        function speakText() {
            const text = document.getElementById('story-content').innerText;
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.lang = currentLang === 'zh' ? 'zh-TW' : 'en-US';
            window.speechSynthesis.speak(utterance);
        }

        // 初始化第一頁
        renderPage();
    </script>
